def run(cmd, cwd):
    t = time.time()
    r = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, timeout=900)
    out = r.stdout + r.stderr
    fails = [l for l in out.splitlines() if l.startswith('# fail')]
    return r.returncode, (fails[-1] if fails else '?'), round(time.time() - t)

results = []
only = set(sys.argv[2:])
for name, f, old, new in M:
    if only and name.split()[0] not in only: continue
    src = f.read_text()
    if src.count(old) < 1:
        results.append({'m': name, 'error': f'pattern count {src.count(old)}'}); print(name, 'PATTERN MISMATCH', src.count(old), flush=True); continue
    f.write_text(src.replace(old, new))  # all occurrences: prover and verifier mutated consistently
    try:
        rc1, f1, t1 = run(['node', '--test'] + [str(p) for p in sorted((D/'packages/crypto/test').glob('*.test.ts'))], D/'packages/crypto')
        rc2, f2, t2 = run(PRO, D/'packages/protocol')
    finally:
        f.write_text(src)
    res = {'m': name, 'crypto': 'KILLED' if rc1 else 'survived', 'crypto_fail': f1, 'protocol': 'KILLED' if rc2 else 'survived', 'protocol_fail': f2, 'secs': t1 + t2}
    results.append(res)
    print(f"{name:55} crypto={res['crypto']:8} ({f1})  protocol-subset={res['protocol']:8} ({f2})  {t1+t2}s", flush=True)
json.dump(results, open(D / ('mutation-results-%s.json' % ('-'.join(sorted(only)) if only else 'all')), 'w'), indent=2)
