const yesNo = (rule: ManifestContest['rule'], tb: ManifestContest['tie_break']): ManifestContest => ({
  id: 'budget', title: 'Budget?', method: 'yes-no', options: [{ id: 'yes', label: 'Yes' }, { id: 'no', label: 'No' }],
  selection_limit: 1, min_selections: 0, rule: rule!, tie_break: tb,
});
const simpleMajority = yesNo({ type: 'threshold', num: 1, den: 2, basis: 'yes+no', strict: true }, { type: 'status-quo', option: 'no' });
const base = (seed: string, extra: Partial<SimOpts> = {}): SimOpts => ({ seed, n: 3, t: 2, voters: 4, contests: [simpleMajority], ...extra });

const results: Record<string, unknown> = {};
async function finish(name: string, sim: Sim, note: string, extra: (b: any) => void = () => {}): Promise<VerificationReport> {
  const b = bundleOf(sim) as any;
  extra(b);
  writeFileSync(join(out, `${name}.bundle.json`), JSON.stringify(b));
  const r = await verifyElection(JSON.parse(JSON.stringify(b)));
  const failed = Object.entries(r.checks).filter(([, s]) => s === 'fail').map(([k]) => k);
  results[name] = {
    note, verdict: r.verdict, failed, counted: r.counts.counted,
    findings: r.findings.filter((f) => f.cls !== 'I').map((f) => `${f.code}/${f.cls}${f.seq === undefined ? '' : `@${f.seq}`}`),
    results: r.results,
  };
  console.log(name.padEnd(28), r.verdict.padEnd(10), JSON.stringify(failed), JSON.stringify((results[name] as any).findings));
  return r;
}

// E1: QV-16-REQ-015. max_spoils_per_voter = 0: the voter publishes a revealed spoil of ballot X, then casts the SAME ballot X.
async function spoilThenCast(name: string, maxSpoils: number): Promise<void> {
  const s = await setup(base(`b1-${name}`, { maxSpoils }));
  const p = s.params!;
  const v = s.voters[0]!;
  const b = buildBallot(p, v.secretKey, v.publicKey, NO_SUPERSEDES, { budget: ['yes'] }, s.rng, 0);
  if (!b.ok) throw new Error('build');
  push(s, makeSpoil(p, v.secretKey, b.value, 'publish')); // reveal (xi + selections) published on the log
  push(s, makeCast(p, v.secretKey, b.value)); // same body, same ballot code, now cast
  s.lastCode.set(H(v.publicKey), H(b.value.code));
  castBallot(s, 1, { budget: ['no'] }); castBallot(s, 2, { budget: ['no'] }); castBallot(s, 3, { budget: ['yes'] });
  closeElection(s); publishTally(s, [1, 2]);
  await finish(name, s, `spoil(publish) then cast of the same code; max_spoils_per_voter=${maxSpoils}`);
}
await spoilThenCast('E1-spoil-over-limit', 0);
