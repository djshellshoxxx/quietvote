// B1 audit PoC (2026-10-28): a signature with R = identity (small order) made by the key holder: S = k*a. Checks which verifiers accept it.
import { ed25519 } from '@noble/curves/ed25519.js';
import { sha512 } from '@noble/hashes/sha2.js';
import { concatBytes, ed25519Verify, leToBigint, sha256, utf8, bytesToHex } from '../packages/crypto/src/index.ts';
const P = ed25519.Point, L = P.Fn.ORDER;
const seed = sha256(utf8('b1-smallR'));
const h = sha512(seed); const a8 = h.slice(0, 32); a8[0] &= 248; a8[31] &= 127; a8[31] |= 64;
const a = leToBigint(a8) % L;
const A = P.BASE.multiply(a).toBytes();
const Rb = P.ZERO.toBytes(); // identity, small order
const msg = utf8('b1 small-order R');
const k = leToBigint(sha512(concatBytes(Rb, A, msg))) % L;
const S = (k * a) % L; const Sb = new Uint8Array(32); let x = S; for (let i = 0; i < 32; i++, x >>= 8n) Sb[i] = Number(x & 0xffn);
const sig = concatBytes(Rb, Sb);
console.log(JSON.stringify({ public_hex: bytesToHex(A), msg_hex: bytesToHex(msg), sig_hex: bytesToHex(sig),
  quietvote_ed25519Verify: ed25519Verify(A, sig, msg), noble_verify_zip215_false: ed25519.verify(sig, msg, A, { zip215: false }) }));
