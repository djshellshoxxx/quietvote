// B3 audit PoC (Playwright, throwaway evidence): observer-facing pages, results, supply-chain (SRI) and automated axe scan.
// Run from the repo root: node packages/web/audit-b3/ui-observer.poc.ts      PASS = the behaviour named in the title was observed.
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { verifyElectionText } from '@quietvote/protocol';
import { FakeBoard, TEST_CONTESTS } from '../test/helpers/fake-board.ts';
import { bundleOf, castBallot, closeElection, publishTally, setup } from '../../protocol/test/helpers/sim.ts';
import { AXE, exe, pw, results, startStack, step } from './harness.ts';

// A finished election: 4 ballots, close, tally published by 2 of 3 trustees.
const sim = await setup({ seed: 'b3-observer', n: 3, t: 2, voters: 4, contests: TEST_CONTESTS()!, minCounted: 3 });
for (let i = 0; i < 4; i++) castBallot(sim, i, { budget: [i < 3 ? 'yes' : 'no'], board: ['p', 'q'] });
closeElection(sim);
publishTally(sim, [1, 2]);
const bundle = bundleOf(sim);
const bundleText = JSON.stringify(bundle);
const rep = await verifyElectionText(bundleText);
console.log('node verifier on the honest bundle: verdict', rep.verdict);
const board = new FakeBoard(sim);
const stack = await startStack(board, sim.voters[0]!.secretKey);
const bundleFile = join(stack.dist, 'bundle.json');
writeFileSync(bundleFile, bundleText);
const base = `${stack.origin}/e/${board.apiId}`;
const browser = await pw.chromium.launch({ executablePath: exe, args: ['--no-sandbox'] });
const text = async (p: import('playwright').Page, sel = 'main'): Promise<string> => (await p.locator(sel).innerText()).replace(/\n+/g, ' | ');
const WRONG_ANCHOR = 'AAAA-AAAA-AAAA-AAAA-AAAA';

try {
  await step('B3-F23 /verify with a mismatching Election ID still prints the positive verdict heading', async () => {
    const p = await (await browser.newContext()).newPage();
    await p.goto(`${stack.origin}/verify`);
    await p.getByLabel('Election bundle file (JSON)').setInputFiles(bundleFile);
    await p.getByLabel(/Election ID you were told/).fill(WRONG_ANCHOR);
    await p.getByRole('button', { name: 'Run checks' }).click();
    await p.locator('main h2').nth(1).waitFor({ timeout: 60000 }).catch(() => undefined);
    await p.waitForTimeout(1500);
    const t = await text(p);
    await p.context().close();
    const shown = /All checks passed|A check failed|Incomplete/.exec(t)?.[0];
    if (!t.includes("doesn't match the Election ID you entered") || shown !== 'All checks passed') throw new Error(t);
    return `the page shows the anchor error AND the heading "${shown}"\n${t.slice(0, 600)}`;
  });

  await step('B3-F23b standalone verifier.html (offline page) behaves the same', async () => {
    const p = await (await browser.newContext()).newPage();
    await p.goto(`${stack.origin}/verifier.html`);
    await p.locator('#file').setInputFiles(bundleFile);
    await p.locator('#anchor').fill(WRONG_ANCHOR);
    await p.locator('#run').click();
    await p.locator('#out h2').waitFor({ timeout: 90000 });
    const t = await text(p, '#out');
    const live = await p.locator('#live').innerText();
    await p.context().close();
    if (!t.includes('All checks passed')) throw new Error(t);
    return `#out: ${t.slice(0, 500)}\n#live (screen reader announcement): ${live}`;
  });

  await step('B3-F7b Results page renders a coordinator-signed tally with no verification step (chart + table), no CSP violations', async () => {
    const p = await (await browser.newContext()).newPage();
    const errs: string[] = [];
    p.on('console', (m) => { if (m.type() === 'error' || /Content Security|Trusted Type/i.test(m.text())) errs.push(m.text()); });
    await p.goto(`${base}/results`);
    await p.locator('main table').first().waitFor({ timeout: 30000 });
    const t = await text(p);
    const hasVerify = /verified|checks passed|verifier/i.test(t);
    await p.context().close();
    return `csp/console errors: ${errs.length ? errs.join(' | ') : 'none'}\nresults text: ${t.slice(t.indexOf('Results')).slice(0, 700)}\nmentions of "verified"/"checks passed" on the page: ${/verified|checks passed/i.test(t)}; disclaimer present: ${/Do not rely on this page alone/.test(t)}; hasVerify=${hasVerify}`;
  });

  await step('B3-F24 SRI covers only the entry module: a tampered statically-imported chunk executes, a tampered entry is blocked', async () => {
    const chunk = readFileSync(join(stack.dist, 'index.html'), 'utf8').match(/assets\/app\.[0-9a-f]{12}\.js/)![0];
    const entrySrc = readFileSync(join(stack.dist, chunk), 'utf8');
    const staticChunks = [...entrySrc.matchAll(/from"\.\/([A-Za-z0-9_-]+\.js)"/g)].map((m) => `assets/${m[1]}`);
    const out: string[] = [`entry statically imports: ${staticChunks.join(', ')}`];
    // (1) tamper the DOM helper chunk (executed on every page load)
    stack.tamper.fn = (path, body) => (staticChunks.some((c) => path.endsWith(c)) ? Buffer.concat([body, Buffer.from(';globalThis.__B3_TAMPERED=1;')]) : body);
    let p = await (await browser.newContext()).newPage();
    const errs1: string[] = [];
    p.on('console', (m) => { if (m.type() === 'error') errs1.push(m.text()); });
    await p.goto(stack.origin + '/');
    await p.getByRole('heading', { name: 'Private votes that anyone can check' }).waitFor({ timeout: 20000 });
    const ran = await p.evaluate(() => (globalThis as { __B3_TAMPERED?: number }).__B3_TAMPERED === 1);
    out.push(`tampered static chunk executed: ${ran}; console errors: ${errs1.length}`);
    await p.context().close();
    // (2) tamper the entry itself
    stack.tamper.fn = (path, body) => (path.endsWith(chunk) ? Buffer.concat([body, Buffer.from(';globalThis.__B3_TAMPERED=1;')]) : body);
    p = await (await browser.newContext()).newPage();
    const errs2: string[] = [];
    p.on('console', (m) => { if (m.type() === 'error') errs2.push(m.text()); });
    await p.goto(stack.origin + '/');
    await p.waitForTimeout(1500);
    const ran2 = await p.evaluate(() => (globalThis as { __B3_TAMPERED?: number }).__B3_TAMPERED === 1);
    const body = await p.locator('body').innerText();
    out.push(`tampered entry executed: ${ran2}; page shows: "${body.replace(/\n+/g, ' ').slice(0, 60)}"; console: ${errs2.join(' | ').slice(0, 160)}`);
    await p.context().close();
    stack.tamper.fn = undefined;
    if (!ran || ran2) throw new Error(out.join('\n'));
    return out.join('\n');
  });

  await step('B3-A1 automated axe-core 4.13 scan of pages the repo\'s own a11y run does not cover (information only, not a conformance statement)', async () => {
    const axe = readFileSync(AXE, 'utf8');
    const ctx = await browser.newContext({ bypassCSP: true });
    const p = await ctx.newPage();
    const lines: string[] = [];
    const scan = async (name: string): Promise<void> => {
      await p.evaluate(axe);
      const r = await p.evaluate(async () => {
        const res = await (globalThis as unknown as { axe: { run(): Promise<{ violations: { id: string; impact: string; nodes: { target: unknown[] }[] }[] }> } }).axe.run();
        return res.violations.map((v) => `${v.id}[${v.impact}] x${v.nodes.length}`);
      });
      lines.push(`${name}: ${r.length ? r.join(', ') : 'no violations reported'}`);
    };
    await p.goto(`${base}`); await p.getByRole('heading', { level: 1 }).waitFor({ timeout: 20000 }); await scan('overview /e/<id>');
    await p.goto(`${base}/rules`); await p.getByRole('heading', { level: 1 }).waitFor(); await scan('rules');
    await p.goto(`${base}/key`); await p.getByLabel('Passphrase', { exact: true }).waitFor(); await scan('key (create form)');
    await p.goto(`${base}/results`); await p.locator('main table').first().waitFor({ timeout: 30000 }); await scan('results (tally)');
    await p.goto(`${base}/receipt`); await p.getByRole('heading', { name: 'Check a receipt' }).first().waitFor(); await scan('receipt');
    await p.goto(`${stack.origin}/verify`); await p.getByRole('button', { name: 'Run checks' }).waitFor(); await scan('verify');
    await p.goto(`${stack.origin}/new`); await p.getByRole('button', { name: 'Freeze election' }).waitFor(); await scan('create election');
    await p.goto(`${stack.origin}/verifier.html`); await scan('verifier.html (standalone)');
    await ctx.close();
    return lines.join('\n');
  });
} finally {
  await browser.close();
  stack.close();
}
const failed = results.filter((r) => !r.ok).length;
console.log(`\n${results.length - failed}/${results.length} steps reproduced`);
process.exit(failed > 0 ? 1 : 0);
