// B3 audit PoC: what a voter gets when selecting the Election ID with the mouse (QV-13 C-CODE: "selectable").
import { FakeBoard, sharedSim } from '../test/helpers/fake-board.ts';
import { exe, pw, startStack } from './harness.ts';
const sim = await sharedSim();
const board = new FakeBoard(sim);
const stack = await startStack(board, sim.voters[0]!.secretKey);
const browser = await pw.chromium.launch({ executablePath: exe, args: ['--no-sandbox'] });
const p = await (await browser.newContext()).newPage();
await p.goto(`${stack.origin}/e/${board.apiId}`);
await p.locator('bdi.code').first().waitFor({ timeout: 20000 });
const sel = await p.evaluate(() => { const el = document.querySelector('bdi.code')!; const r = document.createRange(); r.selectNodeContents(el); const s = getSelection()!; s.removeAllRanges(); s.addRange(r); return s.toString(); });
console.log('selection text :', JSON.stringify(sel));
console.log('copy button text would be the clean ID; manual selection is garbled by the screen-reader-only duplicate.');
await browser.close();
stack.close();
