// B3 audit PoC: time of the in-browser verifier path (protocol.verifyElectionText, main thread) per ballot, to compare with QV-19 s7 "10,000 ballots < 5 min in a Worker pool".
import { verifyElectionText } from '@quietvote/protocol';
import { bundleOf, castBallot, closeElection, publishTally, setup } from '../../protocol/test/helpers/sim.ts';
import { TEST_CONTESTS } from '../test/helpers/fake-board.ts';
for (const voters of [4, 16]) {
  const sim = await setup({ seed: `b3-perf-${voters}`, n: 3, t: 2, voters, contests: TEST_CONTESTS()!, minCounted: 3 });
  for (let i = 0; i < voters; i++) castBallot(sim, i, { budget: ['yes'], board: ['p', 'q'] });
  closeElection(sim); publishTally(sim, [1, 2]);
  const text = JSON.stringify(bundleOf(sim));
  const t0 = performance.now();
  const r = await verifyElectionText(text);
  const ms = performance.now() - t0;
  console.log(`${voters} ballots, 2 contests: verdict ${r.verdict}, ${Math.round(ms)} ms total, ${(ms / voters).toFixed(0)} ms/ballot -> 10,000 ballots ~ ${Math.round((ms / voters) * 10000 / 1000)} s single-threaded (bundle ${(text.length / 1024).toFixed(0)} KiB)`);
}
