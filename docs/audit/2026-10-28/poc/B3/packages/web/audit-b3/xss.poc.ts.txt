// B3 audit PoC (negative result): hostile election text (HTML, script, ICU braces, javascript: URLs) rendered by the SPA.
import { FakeBoard } from '../test/helpers/fake-board.ts';
import { castBallot, closeElection, defaultContests, publishTally, setup } from '../../protocol/test/helpers/sim.ts';
import { PASS, exe, pw, startStack } from './harness.ts';

const evil = [
  '"><img src=x onerror="window.__xss=1">', '<script>window.__xss=1</script>', 'javascript:window.__xss=1', "{t} {n, plural, other {x}} ' '' #", '&lt;b&gt;bold&lt;/b&gt; <b>bold</b>', '<svg onload="window.__xss=1">',
];
const c = defaultContests();
const contests = [{ ...c[0]!, title: evil[0]!, options: [{ id: 'yes', label: evil[1]! }, { id: 'no', label: evil[2]! }] }, { ...c[3]!, id: 'board', title: evil[3]!, options: [{ id: 'p', label: evil[4]! }, { id: 'q', label: evil[5]! }, { id: 'r', label: 'R' }], selection_limit: 2 }];
const sim = await setup({ seed: 'b3-xss', n: 3, t: 2, voters: 4, contests: contests as never, minCounted: 3 });
sim.manifest.title; // title is not editable after setup(): the manifest title is "Test election b3-xss"
for (let i = 0; i < 4; i++) castBallot(sim, i, { budget: ['yes'], board: ['p', 'q'] });
closeElection(sim); publishTally(sim, [1, 2]);
const board = new FakeBoard(sim);
const stack = await startStack(board, sim.voters[0]!.secretKey);
const browser = await pw.chromium.launch({ executablePath: exe, args: ['--no-sandbox'] });
const p = await (await browser.newContext()).newPage();
const errs: string[] = [];
p.on('pageerror', (e) => errs.push(e.message));
p.on('console', (m) => { if (/Content Security|Trusted Type/i.test(m.text())) errs.push(m.text().slice(0, 160)); });
p.on('dialog', async (d) => { errs.push('DIALOG ' + d.message()); await d.dismiss(); });
const check = async (name: string): Promise<void> => {
  const r = await p.evaluate(() => ({ xss: (window as unknown as { __xss?: number }).__xss ?? 0, imgs: document.querySelectorAll('main img, main svg[onload], main script, main b').length, onattrs: document.querySelectorAll('[onerror],[onload],[onclick]').length }));
  console.log(name.padEnd(20), JSON.stringify(r));
};
await p.goto(`${stack.origin}/e/${board.apiId}/results`);
await p.locator('main table').first().waitFor({ timeout: 30000 });
await check('results');
const t = await p.locator('main').innerText();
console.log('literal text shown in results:', ['<script>', 'javascript:', '<svg onload', '{t} {n, plural'].map((x) => `${x} ${t.includes(x) ? 'yes' : 'NO'}`).join(' | '));
await p.goto(`${stack.origin}/e/${board.apiId}/key`);
await p.getByLabel('Use a key file I saved before').check();
await p.getByLabel('Key file', { exact: true }).setInputFiles(stack.keyFile);
await p.getByLabel('Passphrase', { exact: true }).fill(PASS);
await p.getByRole('button', { name: 'Create key' }).click();
await p.getByText('Your key is registered.').waitFor({ timeout: 30000 });
await p.getByRole('link', { name: 'Vote' }).first().click();
await p.locator('main fieldset').first().waitFor();
await check('vote form');
console.log('document.title:', await p.title());
console.log('violations/errors:', errs.join(' | ') || 'none');
await browser.close();
stack.close();
