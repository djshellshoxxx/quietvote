// B3 audit PoC: does the emitted verifier.html work under the _headers file that the same build emits (/* rule)?
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { FakeBoard, TEST_CONTESTS } from '../test/helpers/fake-board.ts';
import { bundleOf, castBallot, closeElection, publishTally, setup } from '../../protocol/test/helpers/sim.ts';
import { exe, pw, startStack } from './harness.ts';

const sim = await setup({ seed: 'b3-vh', n: 3, t: 2, voters: 4, contests: TEST_CONTESTS()!, minCounted: 3 });
for (let i = 0; i < 4; i++) castBallot(sim, i, { budget: ['yes'], board: ['p', 'q'] });
closeElection(sim); publishTally(sim, [1, 2]);
const stack = await startStack(new FakeBoard(sim), sim.voters[0]!.secretKey);
const bundleFile = join(stack.dist, 'bundle.json');
writeFileSync(bundleFile, JSON.stringify(bundleOf(sim)));
const headers = readFileSync(join(stack.dist, '_headers'), 'utf8');
const star = headers.split('\n/assets/*')[0]!.split('\n').filter((l) => l.startsWith('  ')).map((l) => l.trim());
const hdr: Record<string, string> = {};
for (const l of star) { const i = l.indexOf(':'); hdr[l.slice(0, i)] = l.slice(i + 1).trim(); }
const html = readFileSync(join(stack.dist, 'verifier.html'));
const browser = await pw.chromium.launch({ executablePath: exe, args: ['--no-sandbox'] });
for (const [label, extra] of [['meta CSP only (file:// or a static host with no header rule)', {}], ['_headers "/*" rule applied (Netlify / Cloudflare Pages style)', hdr]] as const) {
  const ctx = await browser.newContext();
  const p = await ctx.newPage();
  const msgs: string[] = [];
  p.on('console', (m) => { if (m.type() === 'error') msgs.push(m.text().slice(0, 150)); });
  await p.route('http://verifier.test/**', (r) => r.fulfill({ status: 200, contentType: 'text/html', headers: { ...extra }, body: html }));
  await p.goto('http://verifier.test/verifier.html');
  await p.locator('#file').setInputFiles(bundleFile);
  await p.locator('#run').click();
  let out = '';
  try { await p.locator('#out h2').waitFor({ timeout: 60000 }); out = (await p.locator('#out').innerText()).replace(/\n+/g, ' | ').slice(0, 120); } catch { out = '(no output after 60 s)'; }
  console.log(`${label}\n   result: ${out}\n   console errors: ${msgs.join(' || ') || 'none'}`);
  await ctx.close();
}
await browser.close();
stack.close();
