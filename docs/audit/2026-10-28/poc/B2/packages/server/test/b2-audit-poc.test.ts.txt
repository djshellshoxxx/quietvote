test('B2-03 PoC: an instance_admin key removed from adminKeys keeps full admin power after restart and can take over any org', async (t) => {
  const dir = mkdtempSync(join(tmpdir(), 'qv-b2-admin-'));
  const admin = kp('b2', 'admin');
  const owner = kp('b2', 'owner');
  const attacker = kp('b2', 'attacker');
  const clock = Date.UTC(2026, 9, 28, 12);
  const boot = async (adminKeys: string[]) => {
    const s = buildServer({ dataDir: dir, now: () => clock, logSink: () => {}, minFreeBytes: 0, adminKeys });
    const a = await s.start();
    const inst = (await (await fetch(`${a.url}/api/v1/instance`)).json()) as { instance_id: string };
    const audience = `qv:${inst.instance_id}@http://localhost`;
    let n = 0;
    const call = async (method: string, path: string, body: unknown, key: Ed25519Keypair, idem = false) => {
      const bytes = body === undefined ? new Uint8Array(0) : utf8(JSON.stringify(body));
      const nonce = utf8(`nonce-${adminKeys.length}-${++n}-pad`);
      const headers: Record<string, string> = { authorization: signRequest(key.secretKey, key.publicKey, method, path, audience, bytes, Math.floor(clock / 1000), nonce) };
      if (bytes.length > 0) headers['content-type'] = 'application/json';
      if (idem) headers['idempotency-key'] = `idem-${adminKeys.length}-${n}`;
      const r = await fetch(a.url + path, { method, headers, ...(bytes.length > 0 ? { body: Buffer.from(bytes) } : {}) });
      const text = await r.text();
      return { status: r.status, text, json: text ? (JSON.parse(text) as any) : undefined };
    };
    return { s, call };
  };
  try {
    // boot 1: `admin` is configured; an unrelated org is registered by its owner
    const b1 = await boot([H(admin.publicKey)]);
    assert.equal((await b1.call('GET', '/api/v1/admin/status', undefined, admin)).status, 200);
    const org = await b1.call('POST', '/api/v1/orgs', { name: 'Synthetic Org', owner_key: { pk: b64u(owner.publicKey), alg: 'ed25519' } }, owner, true);
    assert.equal(org.status, 201, org.text);
    const orgId = org.json.org_id as string;
    await b1.s.stop();

    // boot 2: the operator REMOVED the key from the configuration (e.g. after a suspected compromise)
    const b2 = await boot([]);
    try {
      const st = await b2.call('GET', '/api/v1/admin/status', undefined, admin);
      assert.equal(st.status, 200, 'BUG: de-configured admin key is still an instance_admin');
      // ... and it can mint an owner key for an org it does not belong to (cross-org takeover)
      const p = b64u(ed25519Sign(attacker.secretKey, new Uint8Array([...utf8(`QV-KEY-v1\0${orgId}\0`), ...attacker.publicKey])));
      const add = await b2.call('POST', `/api/v1/orgs/${orgId}/keys`, { pk: b64u(attacker.publicKey), alg: 'ed25519', role: 'org_owner', pop: p }, admin);
      assert.equal(add.status, 201, add.text);
      const keys = await b2.call('GET', `/api/v1/orgs/${orgId}/keys`, undefined, attacker);
      assert.equal(keys.status, 200);
      assert.equal((keys.json.items as { role: string }[]).filter((k) => k.role === 'org_owner').length, 2);
      t.diagnostic(`after removal from adminKeys: GET /admin/status -> ${st.status}; POST /orgs/${orgId}/keys (org_owner) -> ${add.status}`);
    } finally {
      await b2.s.stop();
    }
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }

  const w = new World({ seed: 'b2-ttl', n: 3, t: 2, voters: 3 });

    await w.setupOrgAndElection();

    const codes = (inv.json.invites as { code: string }[]).map((x) => x.code);
    const [v0, v1, v2] = [w.voters[0]!, w.voters[1]!, w.voters[2]!];

    // oracle: an unknown code and a used code answer differently (QV-04 §5.1/REQ-004 asks for uniform responses)

    assert.equal(used.status, 409);
    assert.notEqual(unknown.json.code, used.json.code, `unknown=${unknown.json.code} used=${used.json.code}`);
    // no default lifetime: 400 days later the invite still registers (QV-04 default TTL 336 h; QV-29-REQ-008 <= 30 days)

    assert.equal((await register(w, codes[1]!, v1)).status, 201, 'BUG: invite without expires_at is valid forever');
    // QV-09 §4.4: a used code with a DIFFERENT pk must be 409 INVITE_USED; here v0's key (registered with codes[0]) + codes[1] -> 201
    const cross = await register(w, codes[1]!, v0);
    assert.equal(cross.status, 201, `QV-09 §4.4 expects 409 INVITE_USED, got ${cross.status}`);
    void v2;
    t.diagnostic(`unknown -> ${unknown.status} ${unknown.json.code}; used -> ${used.status} ${used.json.code}; +400d -> 201; cross-invite replay -> ${cross.status}`);

      const v = verifyReceipt(rec.value, new Uint8Array(logPk));
      const zero = (await import('@quietvote/crypto')).ed25519Keypair(new Uint8Array(32)).publicKey;
      const vz = verifyReceipt(rec.value, zero);
      t.diagnostic(`receipt from the mirror: verifies under the election log key: ${v.ok}; verifies under the zero-seed key: ${vz.ok}`);
      assert.equal(v.ok, false, 'BUG: receipt does not verify under the log key');
      assert.equal(vz.ok, true, 'it was signed with the all-zero seed placeholder key');
    } finally {
      await B.stop();
    }
  } finally {
    await A.stop();
  }
});

test('B2-07 PoC: restoring an older election DB (e.g. an /admin/backups copy) is not detected; the server then signs a second history (log-key fork)', { timeout: 120_000 }, async (t) => {
  const w = new World({ seed: 'b2-restore', n: 3, t: 2, voters: 3 });
  await w.start();
  try {
    await w.setupOrgAndElection();
    await w.registerVoters();
    assert.equal((await w.publishManifest()).status, 201);
    const db = join(w.dir, 'elections', w.electionId.slice(0, 2), `${w.electionId}.sqlite`);
    const backup = join(w.dir, 'b2-backup.sqlite');
    await w.server.stop(); // closes the DBs (WAL checkpointed)
    copyFileSync(db, backup); // the operator's backup at tree size 1
    await w.start();
    assert.equal((await w.post('ceremony_start', w.coord[0]!, w.marker('dkg-open', w.mh))).status, 201);
    const n1 = (await w.get(`/api/v1/elections/${w.electionId}/log/checkpoint`)).text; // size 2, handed to clients/witnesses
    await w.server.stop();
    copyFileSync(backup, db); // "restore from backup"
    rmSync(`${db}-wal`, { force: true });
    rmSync(`${db}-shm`, { force: true });
    await w.start();
    const st = await w.get(`/api/v1/elections/${w.electionId}/state`);
    assert.equal(st.json.integrity, 'ok', 'no quarantine, no alert after the restore');

    const fork = checkFork(n1, n2, Buffer.from(w.logPkHex, 'hex'));
    t.diagnostic(`restored DB accepted (integrity=${st.json.integrity}); second checkpoint for size 2 signed; checkFork ok=${fork.ok}`);
    assert.equal(fork.ok, true, 'BUG: the log key signed two different roots for size 2');
  } finally {
    await w.stop();
  }
});

test('KNOWN SRV-001 (PR #9 unmerged) still present: a refused request whose peer disconnects kills the whole server process', { timeout: 60_000 }, async (t) => {
  const child = spawn(process.execPath, [join(import.meta.dirname, 'b2-fixtures', 'srv001-child.ts')], { stdio: ['ignore', 'pipe', 'pipe'] });
  let stderr = '';
  child.stderr.on('data', (d) => (stderr += d));
  const exited = new Promise<number | null>((resolve) => child.on('exit', (code) => resolve(code)));
  const port = await new Promise<number>((resolve, reject) => {
    let out = '';
    child.stdout.on('data', (d) => {
      out += d;
      const m = /PORT (\d+)/.exec(out);
      if (m) resolve(Number(m[1]));
    });
    child.on('exit', () => reject(new Error(`child exited early: ${stderr}`)));
  });
  try {
    await new Promise<void>((resolve) => {
      const s = connect(port, '127.0.0.1', () => {
        s.write('POST /api/v1/orgs HTTP/1.1\r\nHost: x\r\nContent-Type: application/json\r\nTransfer-Encoding: chunked\r\n\r\n');
        const chunk = 'x'.repeat(2000);
        for (let i = 0; i < 20; i++) s.write(`${chunk.length.toString(16)}\r\n${chunk}\r\n`);
      });
      s.on('close', () => resolve());
      s.on('error', () => resolve());
      setTimeout(() => s.destroy(), 3000).unref();
    });
    const code = await Promise.race([exited, new Promise<'alive'>((r) => setTimeout(() => r('alive'), 5000))]);
    t.diagnostic(`server process after one unauthenticated request: exit=${String(code)}; stderr: ${stderr.split('\n').find((l) => /TypeError/.test(l)) ?? '(none)'}`);
    assert.notEqual(code, 'alive', 'expected the SRV-001 crash at this SHA');
    assert.match(stderr, /Cannot read properties of null \(reading 'destroy'\)/);
  } finally {
    child.kill('SIGKILL');
  }
});
