const w = new World({ seed: 'explore-wal', n: 3, t: 2, voters: 0, server: { retentionSweepMs: 0 } as never });
await w.start();

const items = Array.from({ length: 5 }, (_, i) => ({ label: `LABELCANARY-${i}-zzzzzzzzzzzz`, contact: `contactcanary-${i}@example.invalid` }));
const inv = (await w.signed('POST', `/api/v1/elections/${w.electionId}/invites`, { items }, w.owner, w.idemKey())).json.invites as { invite_id: string; code: string }[];

const order = [3, 0, 4, 1];
const voters = order.map((i) => kp('explore-wal', `v${i}`));
for (let k = 0; k < order.length; k++) {
  const v = voters[k]!;
  const pop = ed25519Sign(v.secretKey, new Uint8Array([...utf8(`QV-REG-v1\0${w.electionId}\0`), ...v.publicKey]));
  const r = await w.req('POST', `/api/v1/elections/${w.electionId}/registrations`, { invite_code: inv[order[k]!]!.code, pk: b64u(v.publicKey), alg: 'ed25519', pop: b64u(pop) });

  w.tick(60);
}

      if (r.contact_enc && b.includes(Buffer.from(r.contact_enc))) hits.push(`contact_enc#${i}`);
    });
    for (let i = 0; i < 5; i++) { if (b.includes(`LABELCANARY-${i}`)) hits.push(`PLAINTEXT label ${i}`); if (b.includes(`contactcanary-${i}`)) hits.push(`PLAINTEXT contact ${i}`); if (b.includes(inv[i]!.code)) hits.push(`PLAINTEXT code ${i}`); }
    console.log(f.replace(w.dir, ''), b.length, hits.join(' '));
  }
};
