function gateValidate(rt: any) {
  const orig = rt.validate; const origAppend = rt.append;
  let release!: () => void; const released = new Promise<void>((r) => (release = r));
  let arrivals = 0; const waiters: { n: number; r: () => void }[] = [];
  let parkedR!: () => void; const parkedP = new Promise<void>((r) => (parkedR = r));
  rt.append = function (h: Uint8Array, b: Uint8Array) { const p = origAppend.call(rt, h, b); arrivals++; for (const w of waiters) if (arrivals >= w.n) w.r(); return p; };
  rt.validate = async function (h: any, body: Uint8Array) { const v = await orig.call(rt, h, body); parkedR(); await released; return v; };
  return { release, parked: parkedP, arrived: (n: number) => new Promise<void>((r) => { if (arrivals >= n) r(); else waiters.push({ n, r }); }), restore() { delete rt.validate; delete rt.append; } };
}
