function rawReq(path: string, body: unknown): string {
  const b = JSON.stringify(body);
  return `POST ${path} HTTP/1.1\r\nHost: x\r\nContent-Type: application/json\r\nContent-Length: ${Buffer.byteLength(b)}\r\n\r\n${b}`;
}
function pipeline(base: string, reqs: string[]): Promise<{ status: number; body: string }[]> {
  return new Promise((resolve, reject) => {
    const s = connect(Number(new URL(base).port), '127.0.0.1', () => s.write(reqs.join('')));
    let buf = Buffer.alloc(0);
    const out: { status: number; body: string }[] = [];
    const timer = setTimeout(() => { s.destroy(); reject(new Error('timeout ' + out.length)); }, 10000);
    s.on('data', (d) => {
      buf = Buffer.concat([buf, d]);
      for (;;) {
        const i = buf.indexOf('\r\n\r\n');
        if (i < 0) break;
        const head = buf.subarray(0, i).toString();
        const cl = Number(/content-length: (\d+)/i.exec(head)?.[1] ?? 0);
        if (buf.length < i + 4 + cl) break;
        out.push({ status: Number(head.split(' ')[1]), body: buf.subarray(i + 4, i + 4 + cl).toString() });
        buf = buf.subarray(i + 4 + cl);
      }
      if (out.length === reqs.length) { clearTimeout(timer); s.destroy(); resolve(out); }
    });
    s.on('error', reject);
  });
}
