import { verifyBundle, formatReport } from '/home/user/qv-wt/t5-fuzz/packages/verifier/src/index.ts';
const ROOT = '/home/user/qv-wt/t5-fuzz/test-vectors/v1/';

const text = JSON.stringify(full);

}
